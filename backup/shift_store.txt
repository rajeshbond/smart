
func NewStore(db *sql.DB) *Store {
	return &Store{
		db: db,
	}
}

func (s *Store) CreateShift(
	ctx context.Context,
	req CreateShiftRequest,
) (int64, error) {

	var id int64

	err := s.db.QueryRowContext(
		ctx,
		InsertShift,
		req.TenantID,
		req.ShiftCode,
		req.ShiftName,
		req.Description,
		req.CreatedBy,
	).Scan(&id)

	if err != nil {
		return 0, err
	}

	return id, nil
}

func (s *Store) UpdateShift(

func (s *Store) GetAllShifts(
	ctx context.Context,
	tenantID int64,
) ([]ShiftResponse, error) {

	rows, err := s.db.QueryContext(
		ctx,
		GetAllShifts,
		tenantID,
	)

	if err != nil {
		return nil, err
	}

	defer rows.Close()

	var list []ShiftResponse

	for rows.Next() {

		var shift ShiftResponse

		err = rows.Scan(
			&shift.ID,
			&shift.TenantID,
			&shift.ShiftCode,
			&shift.ShiftName,
			&shift.Description,
			&shift.IsActive,
			&shift.CreatedBy,
			&shift.UpdatedBy,
			&shift.CreatedAt,
			&shift.UpdatedAt,
		)

		if err != nil {
			return nil, err
		}

		list = append(list, shift)
	}

	if err = rows.Err(); err != nil {
		return nil, err
	}

	return list, nil
}

func (s *Store) GetCurrentShift(
	ctx context.Context,
	tenantID int64,
	now time.Time,
) (*CurrentShift, error) {

	rows, err := s.db.QueryContext(
		ctx,
		GetShiftSchedules,
		tenantID,
	)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var schedules []shiftSchedule

	for rows.Next() {

		var item shiftSchedule

		err = rows.Scan(
			&item.ShiftMasterID,
			&item.ShiftCode,
			&item.ShiftName,
			&item.StartTime,
			&item.EndTime,
		)

		if err != nil {
			return nil, err
		}

		schedules = append(schedules, item)
	}

	if len(schedules) == 0 {
		return nil, sql.ErrNoRows
	}

	for _, shift := range schedules {

		current, ok := matchShift(now, shift)

		if ok {
			return current, nil
		}
	}

	return nil, sql.ErrNoRows
}
